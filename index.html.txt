// URL Validation
function isValidUrl(string) {
try {
new URL(string.startsWith('http') ? string : 'https://' + string);
return true;
} catch (_) {
return false;
}

// Main Analysis Function
function analyzeURL() {

// Enhanced Phishing Analysis
function performPhishingAnalysis(url) {
const urlObj = new URL(url.startsWith('http') ? url : 'https://' + url);
const hostname = urlObj.hostname.toLowerCase();
const pathname = urlObj.pathname.toLowerCase();

let score = 0;
let checks = [];

// 1. HTTPS
const hasHTTPS = urlObj.protocol === 'https:';
score += hasHTTPS ? 0 : 25;
checks.push({name: 'HTTPS', passed: hasHTTPS, details: hasHTTPS ? 'Encrypted' : 'INSECURE'});

// 2. Keywords
const keywords = ['login','secure','verify','account','bank','paypal','amazon','update'];
const hasKeywords = keywords.some(k => hostname.includes(k));
score += hasKeywords ? 25 : 0;
checks.push({name: 'Keywords', passed: !hasKeywords, details: hasKeywords ? 'PHISHING KEYWORDS' : 'Clean'});

// 3. Homoglyphs
const homoglyphs = /[а-яё]/;
const hasHomoglyphs = homoglyphs.test(hostname);
score += hasHomoglyphs ? 30 : 0;
checks.push({name: 'Homoglyphs', passed: !hasHomoglyphs, details: hasHomoglyphs ? 'FAKE CHARACTERS' : 'Legit'});

// 4. IP Address
const isIP = /^(\d{1,3}\.){3}\d{1,3}$/.test(hostname);
score += isIP ? 25 : 0;
checks.push({name: 'IP Check', passed: !isIP, details: isIP ? 'HIDDEN SERVER' : 'Domain OK'});

// 5. Suspicious Path
const badPaths = ['login','signin','verify','billing','confirm','update'];
const hasBadPath = badPaths.some(p => pathname.includes(p));
score += hasBadPath ? 15 : 0;
checks.push({name: 'Path', passed: !hasBadPath, details: hasBadPath ? 'TRAP PAGE' : 'Normal'});

// Determine risk
const riskLevel = score >= 50 ? 'high' : score >= 25 ? 'medium' : 'low';
const riskLabels = {
high: '🚨 PHISHING DETECTED!',
medium: '⚠️ HIGH RISK',
low: '✅ WEBSITE IS SAFE'
};

return {score: Math.round(score), riskLevel, url, checks, label: riskLabels[riskLevel]};
}

// Display Results
function displayResults(analysis) {
const results = document.getElementById('results');
const scoreDisplay = document.getElementById('scoreDisplay');
const riskLabel = document.getElementById('riskLabel');
const analyzedUrl = document.getElementById('analyzedUrl');
const checksList = document.getElementById('checksList');

scoreDisplay.textContent = analysis.score;
riskLabel.textContent = analysis.label;
analyzedUrl.textContent = analysis.url;

checksList.innerHTML = analysis.checks.map(check => `
               <li style="margin: 10px 0; padding: 10px; background: rgba(0,255,0,0.1); border-radius: 5px;">
                   <strong>${check.name}:</strong> 
                   <span style="color: ${check.passed ? '#00ff00' : '#ff4444'}">${check.details}</span>
               </li>
           `).join('');

results.className = `results risk-${analysis.riskLevel}`;
results.style.display = 'block';

// Auto speak result
setTimeout(speakResult, 500);
}
